   64 |   if (!a || !b || typeof a !== 'string' || typeof b !== 'string') return false;

  165 | /**
  166 |  * Scans a list of prescribed medications for pairwise interactions.
  167 |  */
  168 | export function checkDrugInteractions(medications: Medication[]): DrugInteraction[] {
  169 |   if (!medications || medications.length < 2) return [];
  170 | 
  171 |   const detected: DrugInteraction[] = [];
  172 |   const seenPairs = new Set<string>();
  173 | 
  174 |   for (let i = 0; i < medications.length; i++) {
  175 |     for (let j = i + 1; j < medications.length; j++) {
  176 |       const medA = medications[i];
  177 |       const medB = medications[j];
  178 | 
  179 |       // Combine name + genericName for comprehensive text matching
  180 |       const textA = `${medA.name} ${medA.genericName || ''}`.trim();
  181 |       const textB = `${medB.name} ${medB.genericName || ''}`.trim();
  182 | 
  183 |       if (!textA || !textB) continue;
  184 | 
  185 |       for (const rule of INTERACTION_RULES) {
  186 |         // Special case for duplicate paracetamol: only trigger if two distinct entries match
  187 |         if (rule.id === 'paracetamol-duplicate') {
  188 |           if (rule.matchA.test(textA) && rule.matchB.test(textB)) {
  189 |             const pairKey = `${rule.id}-${i}-${j}`;
  190 |             if (!seenPairs.has(pairKey)) {
  191 |               seenPairs.add(pairKey);
  192 |               detected.push({
  193 |                 id: rule.id,
  194 |                 drugA: medA.name,
  195 |                 drugB: medB.name,
  196 |                 severity: rule.severity,
  197 |                 title: rule.title,
  198 |                 mechanism: rule.mechanism,
  199 |                 recommendation: rule.recommendation,
  200 |               });
  201 |             }
  202 |           }
  203 |           continue;
  204 |         }
  205 | 
  206 |         // Check A -> B or B -> A
  207 |         const matchForward = rule.matchA.test(textA) && rule.matchB.test(textB);
  208 |         const matchReverse = rule.matchA.test(textB) && rule.matchB.test(textA);
  209 | 
  210 |         if (matchForward || matchReverse) {
  211 |           const pairKey = [textA, textB, rule.id].sort().join('::');
  212 |           if (!seenPairs.has(pairKey)) {
  213 |             seenPairs.add(pairKey);
  214 |             detected.push({
  215 |               id: rule.id,
  216 |               drugA: matchForward ? medA.name : medB.name,
  217 |               drugB: matchForward ? medB.name : medA.name,
  218 |               severity: rule.severity,
  219 |               title: rule.title,
  220 |               mechanism: rule.mechanism,
  221 |               recommendation: rule.recommendation,
  222 |             });
  223 |           }
  224 |         }
  225 |       }
  226 |     }
  227 |   }
  228 | 
  229 |   // Sort: CRITICAL first, then HIGH, then MODERATE
  230 |   const order: Record<InteractionSeverity, number> = {
  231 |     CRITICAL: 1,
  232 |     HIGH: 2,
  233 |     MODERATE: 3,
  234 |     CAUTION: 4,
  235 |   };
  236 | 
  237 |   return detected.sort((a, b) => order[a.severity] - order[b.severity]);
  238 | }

  390 | 
  391 |   const patient = await db.query.patients.findFirst({

  398 |   // Prevent deleting patient with active IPD admission

  404 |   }
  405 | 
  406 |   // Cleanly cascade dependent records in an atomic transaction to prevent orphaned rows or FK errors
  407 |   const deleteTx = sqlite.transaction(() => {
